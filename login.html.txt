            </div>

    <script>
        document.getElementById('formLogin').addEventListener('submit', async (e) => {
            e.preventDefault();
            const senha = document.getElementById('senha').value;
            const msgErro = document.getElementById('msgErro');

            try {
                const response = await fetch('https://mes-lavanderia-api.onrender.com/api/login', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ senha })
                });

                const data = await response.json();

                if (response.ok && data.sucesso) {
                    localStorage.setItem('mes_auth_token', data.token);
                    window.location.href = 'dashboard.html';
                } else {
                    msgErro.innerText = data.mensagem || 'Acesso negado!';
                    msgErro.classList.remove('hidden');
                }
            } catch (err) {
                msgErro.innerText = 'Erro ao conectar ao servidor.';
                msgErro.classList.remove('hidden');
            }
        });
    </script>
