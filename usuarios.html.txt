    <script>
        const API_URL = 'https://mes-lavanderia-api.onrender.com';
        let listaUsuariosGlobal = [];

        async function carregarUsuarios() {
            try {
                const res = await fetch(`${API_URL}/api/usuarios`);
                listaUsuariosGlobal = await res.json();
                renderizarTabela();
            } catch (err) {
                console.error("Erro ao carregar usuários:", err);
            }
        }

        function renderizarTabela() {
            const tbody = document.getElementById('tabelaUsuarios');
            if (listaUsuariosGlobal.length === 0) {
                tbody.innerHTML = '<tr><td colspan="4" class="p-4 text-center italic text-slate-400">Nenhum usuário cadastrado.</td></tr>';
                return;
            }

            tbody.innerHTML = listaUsuariosGlobal.map(u => {
                let badgeNivel = '';
                if (u.nivel_acesso === 'admin') badgeNivel = '<span class="bg-purple-100 text-purple-800 px-2 py-0.5 rounded-full font-bold">🟣 Admin</span>';
                else if (u.nivel_acesso === 'supervisor') badgeNivel = '<span class="bg-amber-100 text-amber-800 px-2 py-0.5 rounded-full font-bold">🟡 Supervisor</span>';
                else badgeNivel = '<span class="bg-blue-100 text-blue-800 px-2 py-0.5 rounded-full font-bold">🟢 Operador</span>';

                return `
                    <tr class="hover:bg-slate-50">
                        <td class="p-3 border font-semibold text-slate-800">${u.nome}</td>
                        <td class="p-3 border">${u.usuario}</td>
                        <td class="p-3 border">${badgeNivel}</td>
                        <td class="p-3 border text-center space-x-2">
                            <button onclick="prepararEdicao(${u.id})" class="text-amber-600 font-bold hover:underline">✏️ Editar</button>
                            <button onclick="excluirUsuario(${u.id})" class="text-red-600 font-bold hover:underline">❌ Excluir</button>
                        </td>
                    </tr>
                `;
            }).join('');
        }

        async function salvarUsuario() {
            const id = document.getElementById('usuarioId').value;
            const nome = document.getElementById('nome').value.trim();
            const usuario = document.getElementById('usuario').value.trim();
            const senha = document.getElementById('senha').value;
            const nivelAcesso = document.getElementById('nivelAcesso').value;

            if (!nome || !usuario || (!id && !senha)) {
                alert("Preencha todos os campos obrigatórios!");
                return;
            }

            const method = id ? 'PUT' : 'POST';
            const url = id ? `${API_URL}/api/usuarios/${id}` : `${API_URL}/api/usuarios`;

            try {
                const res = await fetch(url, {
                    method: method,
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ nome, usuario, senha, nivelAcesso })
                });

                if (res.ok) {
                    alert("Usuário salvo com sucesso!");
                    limparFormulario();
                    carregarUsuarios();
                } else {
                    const err = await res.json();
                    alert("Erro: " + (err.mensagem || "Não foi possível salvar"));
                }
            } catch (err) {
                alert("Erro de comunicação com o servidor.");
            }
        }

        function prepararEdicao(id) {
            const u = listaUsuariosGlobal.find(item => item.id === id);
            if (!u) return;

            document.getElementById('usuarioId').value = u.id;
            document.getElementById('nome').value = u.nome;
            document.getElementById('usuario').value = u.usuario;
            document.getElementById('usuario').disabled = true; // Não permite mudar login em edição
            document.getElementById('nivelAcesso').value = u.nivel_acesso;
            document.getElementById('senha').placeholder = "Deixe em branco para manter a atual";

            document.getElementById('tituloForm').innerText = "✏️ Editar Usuário";
        }

        async function excluirUsuario(id) {
            if (!confirm("Tem certeza que deseja excluir este usuário?")) return;

            try {
                const res = await fetch(`${API_URL}/api/usuarios/${id}`, { method: 'DELETE' });
                if (res.ok) {
                    carregarUsuarios();
                } else {
                    alert("Erro ao excluir usuário.");
                }
            } catch (err) {
                alert("Erro de comunicação com o servidor.");
            }
        }

        function limparFormulario() {
            document.getElementById('usuarioId').value = '';
            document.getElementById('nome').value = '';
            document.getElementById('usuario').value = '';
            document.getElementById('usuario').disabled = false;
            document.getElementById('senha').value = '';
            document.getElementById('senha').placeholder = "******";
            document.getElementById('nivelAcesso').value = 'operador';
            document.getElementById('tituloForm').innerText = "➕ Cadastrar Novo Usuário";
        }

        carregarUsuarios();
    </script>
